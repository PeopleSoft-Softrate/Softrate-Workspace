const nodemailer = require('nodemailer');
const { getConnection, decrypt } = require('./email.service');

/**
 * Sends an email using the connected Google Workspace account for the given employee.
 * @param {string} companyCode 
 * @param {string} userId
 * @param {object} mailOptions { to, subject, html, attachments }
 */
async function sendEmailFromCRM(companyCode, userId, mailOptions) {
  // 1. Check if employee has a Google Workspace connection
  const connection = await getConnection(companyCode, userId);
  if (!connection || !connection.connected || !connection.refreshToken) {
    throw new Error('Email sending failed: Google Workspace is not connected for this company.');
  }

  // 2. Decrypt the refresh token securely
  let refreshToken;
  try {
    refreshToken = decrypt(connection.refreshToken);
  } catch (err) {
    throw new Error('Email sending failed: Unable to decrypt Google credentials.');
  }

  // 3. Configure OAuth2 Client
  const { google } = require('googleapis');
  const oauth2Client = new google.auth.OAuth2(
    process.env.GOOGLE_CLIENT_ID,
    process.env.GOOGLE_CLIENT_SECRET,
    process.env.GOOGLE_REDIRECT_URI
  );
  
  oauth2Client.setCredentials({ refresh_token: refreshToken });
  

  // 5. Use Nodemailer to generate the raw MIME message
  const transporter = nodemailer.createTransport({
    streamTransport: true,
    buffer: true
  });

  let fromName = '';
  try {
    const Employee = require('../../../../models/Employee');
    const User = require('../../../../models/User');
    
    const [employee, user] = await Promise.all([
      Employee.findById(userId),
      User.findOne({ companyCode })
    ]);
    
    if (employee) {
      const compName = mailOptions.companyName || (user ? user.companyName : companyCode);
      fromName = `"${employee.name} - ${compName}" `;
    }
  } catch (err) {
    console.warn('[email-sender] Failed to fetch employee name:', err.message);
  }

  const finalMailOptions = {
    ...mailOptions,
    from: fromName ? `${fromName}<${connection.email}>` : connection.email
  };
  console.log('[email-sender] finalMailOptions:', finalMailOptions);

  const messageInfo = await transporter.sendMail(finalMailOptions);
  
  // Gmail API requires base64url format
  const rawMessage = messageInfo.message
    .toString('base64')
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=+$/, '');
  
  console.log('[email-sender] First 200 chars of raw message buffer (decoded):', messageInfo.message.toString('utf8').substring(0, 200));

  // 6. Send the email using Gmail REST API (Bypasses Google SMTP entirely)
  const gmail = google.gmail({ version: 'v1', auth: oauth2Client });
  
  try {
    const res = await gmail.users.messages.send({
      userId: connection.email,
      requestBody: {
        raw: rawMessage
      }
    });
    return { success: true, messageId: res.data.id };
  } catch (err) {
    console.error('[email-sender] Gmail API send failed:', err.message);
    if (err.message.includes('invalid_grant') || err.message.includes('credentials')) {
      throw new Error('Email sending failed: Your Google account connection has expired. Please disconnect and reconnect.');
    }
    throw new Error('Email sending failed: ' + err.message);
  }
}

module.exports = {
  sendEmailFromCRM
};
